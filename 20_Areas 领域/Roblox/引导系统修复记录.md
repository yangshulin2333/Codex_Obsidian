# Roblox 引导系统修复记录

## 问题分析与修复

### 1. `executeStep` 第78行错误
错误原因是 `completeTutorial` 在 `executeStep` 调用时未定义（顺序问题）。已在之前修复中调整了函数顺序。但为了更安全，我们将在 `executeStep` 中增加对步骤和函数的检查。

### 2. 引导启动时机
需要在玩家选主公且传送到主城后，再启动引导，让玩家从主城走向 NPC。

## 修正后的完整代码

### 一、`TutorialController.lua`（修正 executeStep 并增加安全性）

**路径**：`StarterPlayerScripts.Client.Controllers.TutorialController`

```lua
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Workspace = game:GetService("Workspace")
local UserInputService = game:GetService("UserInputService")
local RunService = game:GetService("RunService")
local ContextActionService = game:GetService("ContextActionService")

local RemoteNames = require(ReplicatedStorage.Shared.Constants.RemoteNames)
local TutorialConfig = require(ReplicatedStorage.Shared.Config.TutorialConfig)

local TutorialController = {}
local initialized = false

-- 状态变量
local isRunning = false
local currentStepIndex = 1
local moveConnection = nil
local promptGui = nil
local player = nil
local character = nil
local humanoid = nil
local humanoidRootPart = nil

-- 锁定/解锁控制
local function lockControls()
    UserInputService.MouseBehavior = Enum.MouseBehavior.Default
    ContextActionService:UnbindAllActions()
end

local function unlockControls()
    UserInputService.MouseBehavior = Enum.MouseBehavior.Default
end

-- 创建屏幕提示框
local function showPrompt(text, isFinal)
    if promptGui then promptGui:Destroy() end
    local screenGui = Instance.new("ScreenGui")
    screenGui.Name = "TutorialPromptGui"
    screenGui.ResetOnSpawn = false
    screenGui.Parent = player.PlayerGui

    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(0.8, 0, 0.3, 0)
    frame.Position = UDim2.new(0.1, 0, 0.35, 0)
    frame.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
    frame.BackgroundTransparency = 0.6
    frame.BorderSizePixel = 0
    frame.Parent = screenGui

    local textLabel = Instance.new("TextLabel")
    textLabel.Size = UDim2.new(0.9, 0, 0.7, 0)
    textLabel.Position = UDim2.new(0.05, 0, 0.15, 0)
    textLabel.BackgroundTransparency = 1
    textLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
    textLabel.TextScaled = true
    textLabel.TextWrapped = true
    textLabel.Text = text
    textLabel.Parent = frame

    if not isFinal then
        local button = Instance.new("TextButton")
        button.Size = UDim2.new(0.3, 0, 0.25, 0)
        button.Position = UDim2.new(0.35, 0, 0.7, 0)
        button.BackgroundColor3 = Color3.fromRGB(77, 255, 0)
        button.TextColor3 = Color3.fromRGB(0, 0, 0)
        button.TextScaled = true
        button.Text = "知道了"
        button.Parent = frame
        button.MouseButton1Click:Connect(function()
            screenGui:Destroy()
            promptGui = nil
            -- 继续下一步（如果是等待事件类型，不自动前进）
            if currentStepIndex < #TutorialConfig.Steps then
                local nextStep = TutorialConfig.Steps[currentStepIndex + 1]
                if nextStep.type == "prompt" then
                    executeStep(currentStepIndex + 1)
                end
            end
        end)
    else
        -- 最终完成无按钮，自动结束
        task.delay(3, function()
            screenGui:Destroy()
            promptGui = nil
            completeTutorial()
        end)
    end

    promptGui = screenGui
end

-- 移动角色到目标位置
local function moveToTarget(targetPart, stopDistance, callback)
    if not character or not humanoid or not humanoidRootPart then return end
    local targetPos = targetPart.Position
    if moveConnection then moveConnection:Disconnect() end
    humanoid:MoveTo(targetPos)
    moveConnection = RunService.Heartbeat:Connect(function()
        local distance = (humanoidRootPart.Position - targetPos).Magnitude
        if distance <= stopDistance then
            humanoid:MoveTo(humanoidRootPart.Position) -- 停止移动
            if moveConnection then moveConnection:Disconnect() end
            moveConnection = nil
            if callback then callback() end
        end
    end)
end

-- 完成引导
local function completeTutorial()
    if moveConnection then moveConnection:Disconnect() end
    if promptGui then promptGui:Destroy() end
    unlockControls()
    isRunning = false

    local remotesFolder = ReplicatedStorage:WaitForChild("Remotes")
    local eventsFolder = remotesFolder:FindFirstChild("Events")
    local completeEvent = eventsFolder and eventsFolder:FindFirstChild(RemoteNames.Events.CompleteTutorialRequest)
    if completeEvent then
        completeEvent:FireServer()
        print("[TutorialController] 已通知服务端引导完成")
    else
        warn("[TutorialController] 未找到 CompleteTutorialRequest 事件")
    end
    print("[TutorialController] 引导完成")
end

-- 执行步骤（增加安全检查）
local function executeStep(index)
    if index > #TutorialConfig.Steps then
        completeTutorial()
        return
    end
    currentStepIndex = index
    local step = TutorialConfig.Steps[index]
    if not step then
        warn("[TutorialController] 步骤不存在，索引:", index)
        completeTutorial()
        return
    end

    -- 检查是否为移动步骤
    if step.targetPart then
        local targetModel = Workspace:FindFirstChild("NPC") and Workspace.NPC:FindFirstChild(step.targetPart)
        if targetModel then
            local targetPart = targetModel:FindFirstChild("HumanoidRootPart") or targetModel:FindFirstChild("Torso") or targetModel:FindFirstChildWhichIsA("BasePart")
            if targetPart then
                lockControls()
                moveToTarget(targetPart, step.stopDistance or 10, function()
                    unlockControls()
                    -- 移动完成后，如果下一步是提示，则自动执行
                    if step.nextStep then
                        for i, s in ipairs(TutorialConfig.Steps) do
                            if s.id == step.nextStep then
                                executeStep(i)
                                break
                            end
                        end
                    end
                end)
                return
            end
        end
        warn("[TutorialController] 找不到目标 NPC:", step.targetPart)
        -- 如果找不到目标，跳过此步骤
        executeStep(index + 1)
        return
    end

    -- 提示步骤
    if step.type == "prompt" then
        showPrompt(step.promptText)
        return
    end

    -- 等待事件步骤
    if step.type == "wait_event" then
        -- 不执行任何操作，等待外部事件触发 OnEvent
        return
    end

    -- 完成步骤
    if step.type == "complete" then
        completeTutorial()
        return
    end

    -- 未知类型，跳过
    warn("[TutorialController] 未知步骤类型:", step.type)
    executeStep(index + 1)
end

-- 事件触发接口
function TutorialController.OnEvent(eventName)
    if not isRunning then return end
    local step = TutorialConfig.Steps[currentStepIndex]
    if step and step.type == "wait_event" and step.event == eventName then
        -- 当前步骤等待的事件已发生，进入下一步
        for i, s in ipairs(TutorialConfig.Steps) do
            if s.id == step.nextStep then
                executeStep(i)
                break
            end
        end
    end
end

-- 启动引导
function TutorialController.Start()
    if isRunning then return end
    player = Players.LocalPlayer
    if not player then return end
    character = player.Character
    if not character then
        player.CharacterAdded:Connect(function(ch)
            character = ch
            humanoid = character:FindFirstChild("Humanoid")
            humanoidRootPart = character:FindFirstChild("HumanoidRootPart")
            isRunning = true
            executeStep(1)
        end)
        return
    end
    humanoid = character:FindFirstChild("Humanoid")
    humanoidRootPart = character:FindFirstChild("HumanoidRootPart")
    isRunning = true
    executeStep(1)
end

function TutorialController.Init()
    if initialized then return end
    initialized = true
    player = Players.LocalPlayer

    -- 检查是否已完成引导
    local remotesFolder = ReplicatedStorage:WaitForChild("Remotes")
    local functionsFolder = remotesFolder:FindFirstChild("Functions")
    local checkFunc = functionsFolder and functionsFolder:FindFirstChild("IsTutorialCompleted")
    if checkFunc then
        local success, completed = pcall(checkFunc.InvokeServer, checkFunc)
        if success and completed then
            print("[TutorialController] 引导已完成，跳过")
            return
        end
    end

    print("[TutorialController] 初始化完成，等待触发")
end

return TutorialController
```

### 二、修改 `SelectionController.lua`（选主公并传送完成后启动引导）

**路径**：`StarterPlayerScripts.Client.Controllers.SelectionController`

在选主公成功的响应中，**延迟启动引导**，确保传送已完成。

```lua
-- 在 selectLordEvent 的响应中
selectLordEvent.OnClientEvent:Connect(function(response)
    if response and response.success then
        print("[SelectionController] 选主公成功:", response.faction)
        -- ... 原有界面关闭等代码 ...
        
        -- 延迟启动引导（等待传送和角色稳定）
        task.delay(2, function()
            local TutorialController = require(script.Parent.Parent.Controllers.TutorialController)
            TutorialController.Start()
        end)
    end
end)
```

> **注意**：`TutorialController.Start()` 会检查是否已完成引导，若已完成则不会重复执行。
