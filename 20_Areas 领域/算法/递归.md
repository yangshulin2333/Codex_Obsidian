#### 深拷贝
先看是不是表，不是直接返；

是表就建新表，先登记防循环；

遍历每一对，键值都递归；

最后返回新表。
```lua
local function deepClone(origin, seen)
    -- 处理循环引用
    -- 1. 初始化 seen，如果是第一次调用就创建一个空表
	seen = seen or {}
	-- 2. 检查：这个表是不是已经被拷贝过了？
    if seen[origin] then
        return seen[origin]
    end

    -- 如果不是表，直接返回原值
    if type(origin) ~= "table" then
        return origin
    end

    -- 创建新表
    local copy = {}
    --把原表和复制品绑定记下来
    seen[origin] = copy

    -- 遍历复制
    for k, v in pairs(origin) do
        copy[deepClone(k, seen)] = deepClone(v, seen)
    end

    return copy
end
```





##### C++版
```cpp

#include <iostream>
#include <unordered_map>
// 定义一个嵌套结构体
struct Node {
	int val;
	Node* text;
	//每当创建一个 Node 对象，自动把 val 设为 0,把 next 设为空。
	Node() : val(0),next(nullptr){} 
};
//**& 永远是绑定前面的类型**。
Node* deepClone(Node* origin,std::unordered_map<Node*,Node*>& seen){
	if(!origin) return nullptr;
	//seen.end()代表：查到字典末尾了，**没找到**
	if(seen.find(origin) !=seen.end()){
		return seen[origin];
	}
	//new 类名() **创建对象后，返回的是对象地址**，不是对象本身。
	Node* copy = new Node();
	//把原节点 origin 的 val 值，赋值给新节点 copy 的 val。
	copy->val =origin->val;
	//seen 是一个【存地址→地址】的字典。数据类型就是 std::unordered_map<Node*, Node*>
	seen[origin] = copy;
	copy->next = deepClone(origin->next,seen);
	
	return copy;
}

Node* deepClone(Node* origin){
	std::unordered_map<Node*,Node*> seen;
	return deepClone(origin,seen);
}
```
**注意**：
- 普通对象用 `.` ：`a.val`
- 指针对象用 `->`：`copy->val`、`origin->val`
